CREATE EXTENSION IF NOT EXISTS pgcrypto;

DROP TABLE IF EXISTS produto_v CASCADE;
DROP TABLE IF EXISTS venda CASCADE;
DROP TABLE IF EXISTS produto CASCADE;
DROP TABLE IF EXISTS fornecedor CASCADE;
DROP TABLE IF EXISTS cliente CASCADE;
DROP TABLE IF EXISTS funcionario CASCADE;

CREATE TABLE fornecedor(
    id_f SERIAL PRIMARY KEY,
    nome_f VARCHAR(50) NOT NULL,
    cnpj_f VARCHAR(18),
    tel_f VARCHAR(15),
    data_cad_f DATE
);

CREATE TABLE produto(
    id_p SERIAL PRIMARY KEY,
    nome_p VARCHAR(50) NOT NULL,
    desc_p VARCHAR(100),
    cod_bar_p VARCHAR(20),
    pcusto_p DOUBLE PRECISION,
    pvenda_p DOUBLE PRECISION,
    id_f INTEGER NOT NULL,
    CONSTRAINT produto_fornecedor_fk
        FOREIGN KEY(id_f)
        REFERENCES fornecedor(id_f)
);

CREATE TABLE cliente(
    id_cli SERIAL PRIMARY KEY,
    nome_cli VARCHAR(50) NOT NULL,
    logradouro_cli VARCHAR(50),
    numero_cli INTEGER,
    bairro_cli VARCHAR(30),
    cidade_cli VARCHAR(30),
    estado_cli VARCHAR(2),
    cep_cli VARCHAR(10),
    cpf_cli VARCHAR(14),
    rg_cli VARCHAR(15)
);

CREATE TABLE venda(
    id_v SERIAL PRIMARY KEY,
    dat_v DATE,
    val_v DOUBLE PRECISION,
    id_cli INTEGER NOT NULL,
    CONSTRAINT venda_cliente_fk
        FOREIGN KEY(id_cli)
        REFERENCES cliente(id_cli)
);

CREATE TABLE produto_v(
    id_v INTEGER NOT NULL,
    id_p INTEGER NOT NULL,
    val_p DOUBLE PRECISION,
    qtd_p INTEGER,

    CONSTRAINT produto_v_venda_fk
        FOREIGN KEY(id_v)
        REFERENCES venda(id_v),

    CONSTRAINT produto_v_produto_fk
        FOREIGN KEY(id_p)
        REFERENCES produto(id_p)
);

CREATE TABLE funcionario(
    id_fun SERIAL PRIMARY KEY,
    nome_fun VARCHAR(50) NOT NULL,
    cpf_fun VARCHAR(14),
    login_fun VARCHAR(30) UNIQUE NOT NULL,
    senha_fun VARCHAR(255) NOT NULL,
    tipo_fun VARCHAR(20) NOT NULL
);
